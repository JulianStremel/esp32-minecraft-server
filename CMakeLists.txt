cmake_minimum_required(VERSION 3.16)

# 8/16 denotes PSRAM, independently of the flash size (8 MB minimum).
set(MC_BOARD "esp32s3-8" CACHE STRING "esp32s3-8, esp32s3-16, esp32p4-8, esp32p4-16")
if(NOT MC_BOARD MATCHES "^(esp32s3|esp32p4)-(8|16)$")
    message(FATAL_ERROR "Unsupported MC_BOARD: ${MC_BOARD}")
endif()
set(MC_CHIP "${CMAKE_MATCH_1}")
set(MC_PSRAM_MB "${CMAKE_MATCH_2}")
if(DEFINED IDF_TARGET AND NOT IDF_TARGET STREQUAL MC_CHIP)
    message(FATAL_ERROR "MC_BOARD ${MC_BOARD} requires IDF_TARGET=${MC_CHIP}; use a separate build directory")
endif()
set(IDF_TARGET "${MC_CHIP}" CACHE STRING "ESP-IDF target")
option(MC_EMULATOR "Use reproducible esp-emulator network/server configuration" OFF)
option(MC_BENCH "Run generator checks and the device benchmark instead of the server" OFF)
option(MC_CPU_PROFILE "Log FreeRTOS CPU runtime counters (profiling builds only)" OFF)
option(MC_QEMU_CAPTURE "QEMU S3 profile for GIF capture only" OFF)
if(MC_QEMU_CAPTURE AND (NOT MC_BOARD STREQUAL "esp32s3-8" OR MC_EMULATOR OR MC_BENCH))
    message(FATAL_ERROR "QEMU is only supported for esp32s3-8 GIF capture; use esp-emulator for tests/benchmarks")
endif()
set(MC_NBD_PORT "10809" CACHE STRING "NBD port for the emulator profile")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Per-build sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults.${MC_CHIP}")
if(MC_QEMU_CAPTURE)
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/tools/capture/sdkconfig.defaults")
endif()
if(MC_CPU_PROFILE)
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/tools/emulator/sdkconfig.profile")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/src;${CMAKE_CURRENT_LIST_DIR}/lib/mcore")
set(COMPONENTS src)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
# The component manager records the target in its lockfile. Keep chip switches
# from rewriting a shared tracked lockfile or racing across build directories.
set(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/dependencies.lock.${MC_CHIP}")
idf_build_set_property(DEPENDENCIES_LOCK "${DEPENDENCIES_LOCK}")
project(mcserver)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR IDF_VERSION_MINOR LESS 5)
    message(FATAL_ERROR "Use ESP-IDF >=5.5,<6.0 (tested with v5.5.5)")
endif()
